# /etc/systemd/system/haard-dag.service
[Unit]
Description=Haard: zware klussen overdag
After=docker.service network-online.target
[Service]
User=tim
ExecStart=/home/tim/haard/dagwerk.sh
Nice=19
CPUSchedulingPolicy=idle
IOSchedulingClass=idle

# /etc/systemd/system/haard-dag.timer
[Timer]
OnCalendar=*-*-* 10:30
OnBootSec=2min
[Install]
WantedBy=timers.target

# /etc/systemd/system/haard-stand.service
[Unit]
Description=Haard: dag- of nachtstand van de pc (stil 's nachts)
[Service]
Type=oneshot
ExecStart=/home/tim/haard/stand.sh

# /etc/systemd/system/haard-stand.timer
[Timer]
OnCalendar=*-*-* 23:00
OnCalendar=Mon..Fri *-*-* 08:00
OnCalendar=Sat,Sun *-*-* 10:00
OnBootSec=1min
[Install]
WantedBy=timers.target

# /etc/systemd/system/haard-voorrang.service
[Unit]
Description=Haard: training pauzeren zolang er gesproken wordt
[Service]
ExecStart=/home/tim/haard/voorrang.sh
ExecStopPost=/usr/bin/systemctl kill -s CONT haard-dag.service
Restart=always
[Install]
WantedBy=multi-user.target
